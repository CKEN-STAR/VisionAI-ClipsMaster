```python
class AIViralTransformer:
    def transform_to_viral(subtitles, language, style, intensity):
        # 第1步：叙事结构分析
        narrative_analysis = _analyze_narrative_structure(subtitles, language)
        
        # 第2步：情感强度分析
        emotion_analysis = _analyze_emotion_intensity(subtitles, language)

        return optimized_segments
```

**特点**:
- 5步流程，每步独立
- 依赖外部LLM（`get_llm_for_language`）

- 有丰富的模板库（prefixes, suffixes, intensifiers, hooks）

#### RealAIEngine 架构
```python
class RealAIEngine:
    def generate_viral_subtitle_batch(all_subtitles, language, analysis_results):
        # 第1步：全局理解 - 生成故事摘要
        story_summary = _generate_story_summary(all_subtitles, language, analysis_results)
        
        # 第2步：关键对话提取 - 每集提取30-50%
        all_key_dialogues = []
        for episode_idx, subtitles in enumerate(all_subtitles):
            key_dialogues = _extract_key_dialogues(story_summary, subtitles, episode_idx, language)
            all_key_dialogues.extend(key_dialogues)
        
        # 第3步：分批混剪生成 - 40条一批
        viral_subtitles = _generate_viral_batch(all_key_dialogues, story_summary, language)
        
        # 第4步：连贯性验证
        _validate_batch_coherence(viral_subtitles)
        
        return viral_subtitles
```

**特点**:
- 3步核心流程 + 连贯性验证
- 内置双模型（Qwen2.5 + Mistral）
- 支持GGUF量化和BitsAndBytes量化
- 完整的时间轴映射（保留original_episode, original_index, original_start, original_end）
- 接受外部分析结果（analysis_results参数）

```python
# 添加病毒评分方法（从AIViralTransformer提取）
def calculate_viral_score(self, subtitle: Dict[str, Any], 
                         narrative_analysis: Dict[str, Any],
                         emotion_analysis: Dict[str, Any]) -> float:
    """
    计算字幕的病毒传播潜力评分
    
    Args:
        subtitle: 字幕片段
        narrative_analysis: 叙事分析结果
        emotion_analysis: 情感分析结果
        
    Returns:
        float: 病毒传播评分 (0.0-1.0)
    """
    text = subtitle.get("text", "")
    score = 0.0
    
    # 情感强度权重 (40%)
    emotion_score = self._calculate_emotion_intensity(text)
    score += emotion_score * 0.4
    
    # 关键词匹配权重 (30%)
    keyword_score = self._calculate_keyword_score(text)
    score += keyword_score * 0.3
    
    # 叙事位置权重 (20%)
    narrative_score = self._calculate_narrative_position_score(subtitle, narrative_analysis)
    score += narrative_score * 0.2
    
    # 长度适宜性权重 (10%)
    length_score = self._calculate_length_score(text)
    score += length_score * 0.1
    
    return min(1.0, max(0.0, score))

# 在生成爆款字幕后，为每条字幕添加病毒评分
for subtitle in viral_subtitles:
    subtitle['viral_score'] = self.calculate_viral_score(subtitle, narrative_analysis, emotion_analysis)
```
